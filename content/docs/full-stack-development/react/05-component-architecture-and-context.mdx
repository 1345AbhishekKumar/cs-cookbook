## 1. Lifting State Up

When two or more sibling components need access to the same changing data, lift the state up to their **nearest common ancestor**:

```tsx
// Parent component coordinates shared state
export function Accordion() {
  const [activeId, setActiveId] = useState<string | null>(null);

  return (
    <div>
      <Panel
        title="Account"
        isActive={activeId === "account"}
        onShow={() => setActiveId("account")}
      >
        Account settings...
      </Panel>
      <Panel
        title="Security"
        isActive={activeId === "security"}
        onShow={() => setActiveId("security")}
      >
        Security credentials...
      </Panel>
    </div>
  );
}
```

## 2. Controlled vs Uncontrolled Components

<Tabs items={['Controlled Component', 'Uncontrolled Component']}>
  <Tab value="Controlled Component">
    Form input value is driven strictly by React state:

    ```tsx
    export function ControlledInput() {
      const [email, setEmail] = useState("");

      return (
        <input 
          type="email" 
          value={email} 
          onChange={(e) => setEmail(e.target.value.toLowerCase())} 
        />
      );
    }
    ```
    *Use when you need real-time validation, input masking, or dynamic disabling.*
  </Tab>
  <Tab value="Uncontrolled Component">
    Form input retains its own internal DOM state:

    ```tsx
    export function UncontrolledForm() {
      function handleSubmit(event: React.FormEvent<HTMLFormElement>) {
        event.preventDefault();
        const formData = new FormData(event.currentTarget);
        console.log(formData.get("email"));
      }

      return (
        <form onSubmit={handleSubmit}>
          <input name="email" defaultValue="user@test.com" />
          <button type="submit">Submit</button>
        </form>
      );
    }
    ```
    *Faster, simpler, and standard in React 19 form actions.*
  </Tab>
</Tabs>

## 3. The Context API (React 19 Clean Syntax)

Context provides a way to pass data through the component tree without manually passing props down through every intermediate layer (**Prop Drilling**).

### React 19 Syntax: `<ThemeContext>` without `.Provider`
In React 19, you can render `<ThemeContext value={...}>` directly:

```tsx
import { createContext, useContext, useState } from "react";

type Theme = "light" | "dark";
const ThemeContext = createContext<Theme>("light");

export function App() {
  const [theme, setTheme] = useState<Theme>("dark");

  return (
    // React 19 lets you use <ThemeContext value={...}> directly!
    <ThemeContext value={theme}>
      <Navbar />
      <MainContent />
    </ThemeContext>
  );
}

// Consuming context in deeply nested child:
export function ThemeToggleButton() {
  const theme = useContext(ThemeContext);
  return <span>Current Theme: {theme}</span>;
}
```

## 4. Compound Components Pattern

The **Compound Components** pattern lets you build flexible component APIs where parent and child components communicate implicitly (like native `<select>` and `<option>`):

```tsx
import { createContext, useContext, useState } from "react";

// 1. Internal Context for communication
const TabsContext = createContext<{
  activeTab: string;
  setActiveTab: (id: string) => void;
} | null>(null);

// 2. Parent Container
export function Tabs({ defaultTab, children }: { defaultTab: string; children: React.ReactNode }) {
  const [activeTab, setActiveTab] = useState(defaultTab);
  return (
    <TabsContext value={{ activeTab, setActiveTab }}>
      <div className="tabs-container">{children}</div>
    </TabsContext>
  );
}

// 3. Sub-component: Tab Trigger
Tabs.Trigger = function TabTrigger({ id, children }: { id: string; children: React.ReactNode }) {
  const ctx = useContext(TabsContext);
  if (!ctx) throw new Error("Tabs.Trigger must be inside <Tabs>");

  const isSelected = ctx.activeTab === id;
  return (
    <button
      onClick={() => ctx.setActiveTab(id)}
      className={`tab-btn ${isSelected ? "active" : ""}`}
    >
      {children}
    </button>
  );
};

// 4. Sub-component: Tab Content Panel
Tabs.Content = function TabContent({ id, children }: { id: string; children: React.ReactNode }) {
  const ctx = useContext(TabsContext);
  if (ctx?.activeTab !== id) return null;
  return <div className="tab-panel">{children}</div>;
};
```

### Clean, Declarative Usage:
```tsx
export function SettingsPage() {
  return (
    <Tabs defaultTab="profile">
      <div className="tab-bar">
        <Tabs.Trigger id="profile">Profile</Tabs.Trigger>
        <Tabs.Trigger id="security">Security</Tabs.Trigger>
      </div>

      <Tabs.Content id="profile">Profile Settings Panel</Tabs.Content>
      <Tabs.Content id="security">Security Credentials Panel</Tabs.Content>
    </Tabs>
  );
}
```
