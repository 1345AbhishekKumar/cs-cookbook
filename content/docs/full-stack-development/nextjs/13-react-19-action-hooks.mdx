Next.js is built around React 19, which introduces three foundational hooks for managing asynchronous action lifecycles without manual `useState` or `useEffect` boilerplate.

---

## 1. `useActionState` (Managing Form State & Errors)

`useActionState` manages the state returned by a Server Action (e.g. validation error messages or success status) and tracks execution pending state:

```tsx
'use client';

import { useActionState } from 'react';
import { createFeedback } from '@/app/actions/feedback';

export function FeedbackForm() {
  const [state, formAction, isPending] = useActionState(createFeedback, {
    errors: {},
    message: '',
  });

  return (
    <form action={formAction} className="space-y-4">
      <input type="text" name="name" placeholder="Your name" />
      {state.errors?.name && <p className="text-red-500">{state.errors.name[0]}</p>}

      <textarea name="comment" placeholder="Feedback..." />
      {state.errors?.comment && <p className="text-red-500">{state.errors.comment[0]}</p>}

      <button type="submit" disabled={isPending}>
        {isPending ? 'Submitting...' : 'Send Feedback'}
      </button>

      {state.message && <p className="text-green-500">{state.message}</p>}
    </form>
  );
}
```

## 2. `useFormStatus` (Pending Buttons Without Prop Drilling)

`useFormStatus` gives child components access to the pending status of their parent `<form>`. It eliminates passing `isPending` props down through component trees:

```tsx
'use client';

import { useFormStatus } from 'react-dom';

export function SubmitButton({ label }: { label: string }) {
  const { pending } = useFormStatus();

  return (
    <button
      type="submit"
      disabled={pending}
      className="px-4 py-2 bg-blue-600 text-white rounded disabled:opacity-50"
    >
      {pending ? 'Processing...' : label}
    </button>
  );
}
```

<Callout type="warning" title="Placement Rule for useFormStatus">
`useFormStatus` must be called from a component rendered **inside** a `<form>` tag, not in the same component that renders the `<form>`.
</Callout>

## 3. `useOptimistic` (Zero-Latency UI Feedback)

`useOptimistic` allows you to immediately update the client interface before the server mutation finishes, with automatic rollback if an error occurs:

```tsx
// components/OptimisticTodoList.tsx
'use client';

import { useOptimistic } from 'react';
import { addTodo } from '@/app/actions/todos';
import { SubmitButton } from './SubmitButton';

type Todo = { id: string; title: string };

export function OptimisticTodoList({ initialTodos }: { initialTodos: Todo[] }) {
  const [optimisticTodos, addOptimisticTodo] = useOptimistic(
    initialTodos,
    (state, newTitle: string) => [
      ...state,
      { id: 'optimistic-' + Math.random(), title: newTitle },
    ]
  );

  async function handleAction(formData: FormData) {
    const title = formData.get('title') as string;
    // 1. Immediately render item in client UI
    addOptimisticTodo(title);
    // 2. Transmit mutation to server
    await addTodo(formData);
  }

  return (
    <div>
      <form action={handleAction}>
        <input name="title" required placeholder="New task..." />
        <SubmitButton label="Add Task" />
      </form>

      <ul className="mt-4 space-y-2">
        {optimisticTodos.map((todo) => (
          <li key={todo.id} className="p-2 border rounded">
            {todo.title}
          </li>
        ))}
      </ul>
    </div>
  );
}
```
