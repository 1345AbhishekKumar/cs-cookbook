## Project 1: Elden Ring: The Lands Between Gauntlet

An expanded, multi-phase text adventure RPG featuring classes, consumable flasks, boss battles with Margit and Godrick, and the coffee cheat code!

```bash title="elden_ring_full.sh"
#!/bin/bash
# ==============================================================================
# Elden Ring: The Lands Between Gauntlet (Expanded Edition)
# ==============================================================================
set -e

COLOR_RED="\033[1;31m"
COLOR_GREEN="\033[1;32m"
COLOR_YELLOW="\033[1;33m"
COLOR_CYAN="\033[1;36m"
COLOR_RESET="\033[0m"

echo -e "${COLOR_CYAN}=========================================================="
echo "          ELDEN RING: THE BASH GAUNTLET                   "
echo -e "==========================================================${COLOR_RESET}"

# Class Selection
echo "Choose your hero archetype:"
echo "1) Samurai     (HP: 25, ATK: 12, Flasks: 2)"
echo "2) Astrologer  (HP: 18, ATK: 16, Flasks: 3)"
echo "3) Coffee Mage (HP: 22, ATK: 14, Flasks: 5)"

read -p "Select class [1-3]: " hero_choice

case $hero_choice in
  1)
    class_name="Samurai"; player_hp=25; player_atk=12; flasks=2 ;;
  2)
    class_name="Astrologer"; player_hp=18; player_atk=16; flasks=3 ;;
  3)
    class_name="Coffee Mage"; player_hp=22; player_atk=14; flasks=5 ;;
  *)
    class_name="Wretch"; player_hp=15; player_atk=10; flasks=1 ;;
esac

echo -e "\nYou venture forth as a ${COLOR_YELLOW}$class_name${COLOR_RESET}!"
sleep 1

# Combat Function
fight_boss() {
  local boss_name=$1
  local boss_hp=$2
  local boss_atk=$3

  echo -e "\n${COLOR_RED}⚔️ WARNING: The fog gates open! $boss_name appears! (HP: $boss_hp)${COLOR_RESET}"
  sleep 1

  while [ $boss_hp -gt 0 ] && [ $player_hp -gt 0 ]; do
    echo -e "\n--------------------------------------------------"
    echo -e "Your HP: ${COLOR_GREEN}$player_hp${COLOR_RESET} | Flasks: $flasks | $boss_name HP: ${COLOR_RED}$boss_hp${COLOR_RESET}"
    echo "Actions: 1) Light Attack  2) Heavy Attack  3) Heal  4) Secret Word"
    read -p "Choose action [1-4]: " action

    case $action in
      1)
        # 100% hit chance, base attack
        local dmg=$(( player_atk + (RANDOM % 4) ))
        boss_hp=$(( boss_hp - dmg ))
        echo -e "🗡️ You strike $boss_name for ${COLOR_YELLOW}$dmg${COLOR_RESET} damage!"
        ;;
      2)
        # 50% hit chance, double damage
        if [ $(( RANDOM % 2 )) -eq 0 ]; then
          local dmg=$(( (player_atk * 2) + (RANDOM % 8) ))
          boss_hp=$(( boss_hp - dmg ))
          echo -e "💥 CRITICAL STRIKE! You deal ${COLOR_YELLOW}$dmg${COLOR_RESET} damage to $boss_name!"
        else
          echo "💨 You swung heavily and MISSED!"
        fi
        ;;
      3)
        if [ $flasks -gt 0 ]; then
          player_hp=$(( player_hp + 15 ))
          flasks=$(( flasks - 1 ))
          echo -e "🧪 You chug an Estus Flask! Restored 15 HP! Flasks left: $flasks"
        else
          echo "⚠️ Your flasks are completely empty!"
        fi
        ;;
      4|"coffee")
        echo -e "${COLOR_YELLOW}☕ COFFEE INCANTATION CAST! Golden lightning obliterates the arena!${COLOR_RESET}"
        boss_hp=0
        ;;
      *)
        echo "You hesitated and fumbled your weapon!"
        ;;
    esac

    # Boss Counter-Attack if still alive
    if [ $boss_hp -gt 0 ]; then
      local boss_dmg=$(( (RANDOM % boss_atk) + 3 ))
      player_hp=$(( player_hp - boss_dmg ))
      echo -e "${COLOR_RED}🩸 $boss_name retaliates and hits you for $boss_dmg damage!${COLOR_RESET}"
      sleep 1
    fi
  done

  if [ $player_hp -le 0 ]; then
    echo -e "\n${COLOR_RED}=========================================================="
    echo "                      YOU DIED.                           "
    echo -e "==========================================================${COLOR_RESET}"
    exit 1
  else
    echo -e "\n${COLOR_GREEN}🏆 VICTORY ACHIEVED! $boss_name has been vanquished!${COLOR_RESET}"
  fi
}

# Boss Battles
fight_boss "Margit the Fell Omen" 30 10
echo "You rest briefly by the Site of Grace. Restoring 10 HP..."
player_hp=$(( player_hp + 10 ))

fight_boss "Godrick the Grafted" 50 14

echo -e "\n${COLOR_CYAN}👑 BECOME THE ELDEN LORD! You conquered Stormveil Castle!${COLOR_RESET}"
exit 0
```

## Project 2: Interactive Fitness & Push-Up Coach

A command-line fitness coach that tracks sets, reps, rest intervals, and logs workout data to a persistent CSV database:

```bash title="fitness_coach.sh"
#!/bin/bash
# ==============================================================================
# Terminal Fitness Coach & Push-Up Logger
# ==============================================================================
set -euo pipefail

LOG_FILE="$HOME/.bash_workout_log.csv"

# Initialize CSV log if not present
if [ ! -f "$LOG_FILE" ]; then
  echo "Timestamp,Workout,Sets,RepsPerSet,TotalReps" > "$LOG_FILE"
fi

echo "=========================================================="
echo "          BASH FITNESS COACH & WORKOUT TRACKER            "
echo "=========================================================="

read -p "Workout type (Push-ups / Squats / Pull-ups): " workout_type
read -p "How many sets will you complete today? " total_sets
read -p "How many reps per set? " reps_per_set
read -p "Rest interval between sets in seconds [e.g. 30]: " rest_seconds

current_set=1

while [ $current_set -le $total_sets ]; do
  echo ""
  echo "💪 --- SET $current_set OF $total_sets ---"
  echo "Perform $reps_per_set repetitions now!"
  read -p "Press [Enter] when set $current_set is completed..."

  echo "✅ Set $current_set done!"

  if [ $current_set -lt $total_sets ]; do
    echo -n "Resting for $rest_seconds seconds: "
    for (( s = rest_seconds; s > 0; s-- )); do
      echo -n "$s "
      sleep 1
    done
    echo "⚡ TIME IS UP!"
  fi

  current_set=$(( current_set + 1 ))
done

total_reps=$(( total_sets * reps_per_set ))
timestamp=$(date "+%Y-%m-%d %H:%M:%S")

# Append record to CSV
echo "$timestamp,$workout_type,$total_sets,$reps_per_set,$total_reps" >> "$LOG_FILE"

echo ""
echo "=========================================================="
echo "🎉 WORKOUT FINISHED! Total $workout_type completed: $total_reps"
echo "Saved workout log to $LOG_FILE"
echo "=========================================================="
exit 0
```

## Project 3: Production Automated Backup Engine with Retention

A defensive backup script with `set -euo pipefail`, signal trapping, gzip compression, checksum generation, and automatic retention rotation:

```bash title="backup_engine.sh"
#!/bin/bash
# ==============================================================================
# Enterprise Backup & Retention Engine
# Usage: ./backup_engine.sh /var/www/html /backups/web 7
# ==============================================================================
set -euo pipefail

SOURCE_DIR="${1:-/var/www/html}"
BACKUP_DEST="${2:-/tmp/backups}"
RETENTION_COUNT="${3:-5}"  # Keep last 5 backups

TIMESTAMP=$(date +"%Y%m%d_%H%M%S")
ARCHIVE_NAME="backup_$(basename "$SOURCE_DIR")_${TIMESTAMP}.tar.gz"
TEMP_DIR=$(mktemp -d -t backup_work_XXXXXX)

cleanup() {
  local exit_status=$?
  if [ -d "$TEMP_DIR" ]; then
    rm -rf "$TEMP_DIR"
  fi
  if [ $exit_status -ne 0 ]; then
    echo "❌ [$(date)] Backup job failed with exit status $exit_status" >&2
  fi
}
trap cleanup EXIT SIGINT SIGTERM

echo "📦 [$(date)] Initiating backup for $SOURCE_DIR"
mkdir -p "$BACKUP_DEST"

# 1. Create compressed tarball
tar -czf "$TEMP_DIR/$ARCHIVE_NAME" -C "$(dirname "$SOURCE_DIR")" "$(basename "$SOURCE_DIR")"

# 2. Generate SHA256 checksum for integrity verification
sha256sum "$TEMP_DIR/$ARCHIVE_NAME" > "$TEMP_DIR/$ARCHIVE_NAME.sha256"

# 3. Move archive to final backup directory
mv "$TEMP_DIR/$ARCHIVE_NAME" "$BACKUP_DEST/"
mv "$TEMP_DIR/$ARCHIVE_NAME.sha256" "$BACKUP_DEST/"

echo "✅ [$(date)] Successfully stored $ARCHIVE_NAME in $BACKUP_DEST"

# 4. Prune older backups beyond RETENTION_COUNT
echo "🧹 Checking backup retention (keeping last $RETENTION_COUNT archives)..."
cd "$BACKUP_DEST"
ls -1t backup_*.tar.gz 2>/dev/null | tail -n +$(( RETENTION_COUNT + 1 )) | while read -r old_archive; do
  echo "🗑️ Pruning outdated backup: $old_archive"
  rm -f "$old_archive" "${old_archive}.sha256"
done

echo "✨ [$(date)] Backup and rotation completed cleanly."
exit 0
```

## Project 4: Subnet IP & Port Sweeper

A network scanner in pure Bash that scans a `/24` subnet for responsive hosts and checks open ports (22, 80, 443) using `/dev/tcp`:

```bash title="subnet_scanner.sh"
#!/bin/bash
# ==============================================================================
# Subnet & Port Scanner (Pure Bash)
# Usage: ./subnet_scanner.sh 192.168.1
# ==============================================================================

SUBNET="${1:-192.168.1}"
REPORT_FILE="/tmp/scan_report_${SUBNET}.txt"

echo "=========================================================="
echo "    SUB-NET & PORT SCANNER FOR $SUBNET.0/24               "
echo "=========================================================="
echo "Scan started at $(date)" > "$REPORT_FILE"

check_port() {
  local ip=$1
  local port=$2
  # Timeout connection after 1 second using subshell & redirection
  timeout 1 bash -c "cat < /dev/null > /dev/tcp/$ip/$port" 2>/dev/null && echo "OPEN" || echo "CLOSED"
}

# Scan hosts 1 to 20
for host in {1..20}; do
  ip="$SUBNET.$host"

  # Quick ICMP ping probe
  if ping -c 1 -W 1 "$ip" > /dev/null 2>&1; then
    echo -e "\n🟢 Host UP: $ip"
    echo "Host UP: $ip" >> "$REPORT_FILE"

    # Check common ports
    for port in 22 80 443; do
      status=$(check_port "$ip" "$port")
      if [ "$status" == "OPEN" ]; then
        echo "   └── Port $port: 🟢 OPEN"
        echo "   └── Port $port: OPEN" >> "$REPORT_FILE"
      fi
    done
  fi
done

echo ""
echo "=========================================================="
echo "Scan complete. Full report written to $REPORT_FILE"
cat "$REPORT_FILE"
exit 0
```

## Project 5: Production Server Health Monitor & Alert

A diagnostic monitor that checks CPU load, memory availability, root disk storage, and critical services, alerting in red if thresholds are breached:

```bash title="health_monitor.sh"
#!/bin/bash
# ==============================================================================
# Server Health & Resource Dashboard
# ==============================================================================
set -euo pipefail

COLOR_GREEN="\033[1;32m"
COLOR_YELLOW="\033[1;33m"
COLOR_RED="\033[1;31m"
COLOR_RESET="\033[0m"

# Thresholds
DISK_THRESHOLD=80
MEM_THRESHOLD=85

echo "=========================================================="
echo "           SERVER HEALTH MONITORING DASHBOARD             "
echo "=========================================================="
echo "Hostname : $(hostname)"
echo "Kernel   : $(uname -r)"
echo "Uptime   : $(uptime -p)"
echo "Date     : $(date '+%Y-%m-%d %H:%M:%S')"
echo "=========================================================="

# 1. Disk Usage Check
disk_usage=$(df -h / | awk 'NR==2 {print $5}' | tr -d '%')
echo -n "Disk Usage (Root /): $disk_usage% "
if [ "$disk_usage" -ge "$DISK_THRESHOLD" ]; then
  echo -e "${COLOR_RED}[CRITICAL - Exceeds $DISK_THRESHOLD%]${COLOR_RESET}"
else
  echo -e "${COLOR_GREEN}[HEALTHY]${COLOR_RESET}"
fi

# 2. Memory Usage Check
mem_total=$(free -m | awk '/Mem:/ {print $2}')
mem_used=$(free -m | awk '/Mem:/ {print $3}')
mem_percent=$(( mem_used * 100 / mem_total ))
echo -n "RAM Usage          : $mem_percent% ($mem_used MB / $mem_total MB) "
if [ "$mem_percent" -ge "$MEM_THRESHOLD" ]; then
  echo -e "${COLOR_RED}[WARNING - Exceeds $MEM_THRESHOLD%]${COLOR_RESET}"
else
  echo -e "${COLOR_GREEN}[HEALTHY]${COLOR_RESET}"
fi

# 3. CPU 1-Minute Load
load_1m=$(awk '{print $1}' /proc/loadavg)
echo "CPU Load (1-min)   : $load_1m"

# 4. Critical Services Check
echo -e "\nService Status Checks:"
for svc in ssh cron docker; do
  if systemctl is-active --quiet "$svc" 2>/dev/null; then
    echo -e "   └── $svc: ${COLOR_GREEN}ACTIVE (Running)${COLOR_RESET}"
  else
    echo -e "   └── $svc: ${COLOR_YELLOW}INACTIVE / NOT FOUND${COLOR_RESET}"
  fi
done

echo "=========================================================="
exit 0
```
