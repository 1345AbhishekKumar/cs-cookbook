## The Binomial Coefficient Formula

The number of ways to choose $k$ elements from an $n$-element set is given by:
$$\binom{n}{k} = \frac{n!}{k! \cdot (n - k)!}$$

When computing $\binom{n}{k} \pmod M$ for repeated queries:

1. **Precompute factorials**: $\text{fact}[i] = (i \cdot \text{fact}[i-1]) \pmod M$ in $\mathcal{O}(N)$ time.
2. **Precompute inverse factorials**: $\text{invFact}[i] = (\text{invFact}[i+1] \cdot (i + 1)) \pmod M$ in $\mathcal{O}(N)$ time.
3. **Answer any $\binom{n}{k}$ query in strictly $\mathcal{O}(1)$ time**:
   $$\binom{n}{k} \equiv \text{fact}[n] \cdot \text{invFact}[k] \cdot \text{invFact}[n - k] \pmod M$$

## Canonical Application: Catalan Numbers

The $n$-th **Catalan Number** $C_n$ describes fundamental counting structures:
$$C_n = \frac{1}{n + 1} \binom{2n}{n} = \frac{(2n)!}{(n + 1)! \, n!}$$

- Number of valid expressions containing $n$ pairs of parentheses.
- Number of distinct binary search trees that can be formed using $n$ distinct keys.
- Number of monotonic paths along the edges of an $n \times n$ grid that do not cross above the diagonal.

```python
class Combinatorics:
    def __init__(self, max_n: int, mod: int = 1_000_000_007):
        self.mod = mod
        self.fact = [1] * (max_n + 1)
        self.inv_fact = [1] * (max_n + 1)

        # 1. Precompute factorials
        for i in range(1, max_n + 1):
            self.fact[i] = (self.fact[i - 1] * i) % mod

        # 2. Fermat inverse of max_n!
        self.inv_fact[max_n] = pow(self.fact[max_n], mod - 2, mod)

        # 3. Backward precomputation of inverse factorials
        for i in range(max_n - 1, -1, -1):
            self.inv_fact[i] = (self.inv_fact[i + 1] * (i + 1)) % mod

    def ncr(self, n: int, r: int) -> int:
        """O(1) calculation of nCr mod M."""
        if r < 0 or r > n:
            return 0
        return self.fact[n] * self.inv_fact[r] % self.mod * self.inv_fact[n - r] % self.mod

    def catalan(self, n: int) -> int:
        """O(1) calculation of n-th Catalan number mod M."""
        return (self.ncr(2 * n, n) * pow(n + 1, self.mod - 2, self.mod)) % self.mod
````

```typescript
export class Combinatorics {
  private mod: bigint = 1000000007n;
  private fact: bigint[];
  private invFact: bigint[];

  constructor(maxN: number) {
    this.fact = new Array(maxN + 1).fill(1n);
    this.invFact = new Array(maxN + 1).fill(1n);

    for (let i = 1; i <= maxN; i++) {
      this.fact[i] = (this.fact[i - 1] * BigInt(i)) % this.mod;
    }

    const power = (b: bigint, exp: bigint): bigint => {
      let res = 1n;
      b %= this.mod;
      while (exp > 0n) {
        if ((exp & 1n) === 1n) res = (res * b) % this.mod;
        b = (b * b) % this.mod;
        exp >>= 1n;
      }
      return res;
    };

    this.invFact[maxN] = power(this.fact[maxN], this.mod - 2n);
    for (let i = maxN - 1; i >= 0; i--) {
      this.invFact[i] = (this.invFact[i + 1] * BigInt(i + 1)) % this.mod;
    }
  }

  nCr(n: number, r: number): bigint {
    if (r < 0 || r > n) return 0n;
    return (((this.fact[n] * this.invFact[r]) % this.mod) * this.invFact[n - r]) % this.mod;
  }
}
````
