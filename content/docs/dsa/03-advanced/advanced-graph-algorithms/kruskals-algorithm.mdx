A **Minimum Spanning Tree (MST)** of an undirected, connected, weighted graph $G = (V, E)$ is an acyclic subset of edges $T \subseteq E$ that connects all $V$ vertices with minimum total edge weight:

$$\min \sum_{e \in T} w(e), \quad \text{where } |T| = |V| - 1$$

**Kruskal's Algorithm** is an edge-centric greedy algorithm:

1. Sort all $E$ edges in non-decreasing order of weight.
2. Initialize an empty tree $T$ and a Disjoint Set Union (DSU) data structure.
3. For each edge $(u, v, w)$ in sorted order:
   - If $\text{Find}(u) \ne \text{Find}(v)$, adding this edge does not create a cycle.
   - Include $(u, v)$ in $T$, and invoke $\text{Union}(u, v)$.
4. Terminate when $|T| = V - 1$.

```python
class DSU:
    def __init__(self, n: int):
        self.parent = list(range(n))
        self.rank = [0] * n

    def find(self, i: int) -> int:
        if self.parent[i] != i:
            self.parent[i] = self.find(self.parent[i])
        return self.parent[i]

    def union(self, i: int, j: int) -> bool:
        root_i, root_j = self.find(i), self.find(j)
        if root_i == root_j:
            return False
        if self.rank[root_i] < self.rank[root_j]:
            root_i, root_j = root_j, root_i
        self.parent[root_j] = root_i
        if self.rank[root_i] == self.rank[root_j]:
            self.rank[root_i] += 1
        return True
````

```typescript
export function kruskalMst(
  numNodes: number,
  edges: [number, number, number][]
): { totalCost: number; mstEdges: [number, number, number][] } {
  const sorted = [...edges].sort((a, b) => a[2] - b[2]);
  const parent = Array.from({ length: numNodes }, (_, i) => i);

  function find(i: number): number {
    if (parent[i] !== i) parent[i] = find(parent[i]);
    return parent[i];
  }

  function union(i: number, j: number): boolean {
    const rootI = find(i);
    const rootJ = find(j);
    if (rootI === rootJ) return false;
    parent[rootI] = rootJ;
    return true;
  }

  let totalCost = 0;
  const mstEdges: [number, number, number][] = [];

  for (const [u, v, w] of sorted) {
    if (union(u, v)) {
      mstEdges.push([u, v, w]);
      totalCost += w;
      if (mstEdges.length === numNodes - 1) break;
    }
  }

  return { totalCost, mstEdges };
}
````

std::sort(edges.begin(), edges.end());
std::vector<int> parent(numNodes);
std::iota(parent.begin(), parent.end(), 0);

    auto find = [&](auto& self, int i) -> int {
        return parent[i] == i ? i : (parent[i] = self(self, parent[i]));
    };

    long long totalCost = 0;
    std::vector<Edge> mst;

    for (const auto& edge : edges) {
        int rootU = find(find, edge.u);
        int rootV = find(find, edge.v);
        if (rootU != rootV) {
            parent[rootU] = rootV;
            totalCost += edge.w;
            mst.push_back(edge);
            if (static_cast<int>(mst.size()) == numNodes - 1) break;
        }
    }

    return {totalCost, mst};
