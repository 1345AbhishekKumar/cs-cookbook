An **Eulerian Trail (Path)** is a walk in a finite graph that visits **every edge exactly once**. An **Eulerian Circuit (Cycle)** is an Eulerian trail that starts and ends at the exact same vertex.

This foundational problem, first solved by Leonhard Euler in 1736 to resolve the Seven Bridges of Königsberg, lies at the heart of genome fragment assembly (de Bruijn graphs), CNC milling toolpaths, and snowplow routing.

---

## 1. Existence Theorems

### Undirected Graphs

- An **Eulerian Circuit** exists if and only if:
  1. All vertices with non-zero degree belong to a single connected component.
  2. **Every vertex has an even degree**.
- An **Eulerian Path** exists if and only if:
  1. All edges belong to a single connected component.
  2. **Exactly zero or two vertices have an odd degree**. (If two, they must be the start and end nodes).

### Directed Graphs

- An **Eulerian Circuit** exists if and only if:
  1. All non-isolated vertices belong to a single strongly connected component.
  2. For every vertex: $\text{in-degree}(v) = \text{out-degree}(v)$.
- An **Eulerian Path** exists if and only if:
  - At most one vertex has $\text{out-degree} - \text{in-degree} = 1$ (the start vertex).
  - At most one vertex has $\text{in-degree} - \text{out-degree} = 1$ (the end vertex).
  - All other vertices have $\text{in-degree} = \text{out-degree}$.

---

## 2. Hierholzer's Algorithm

Hierholzer's algorithm finds an Eulerian circuit in linear **$\mathcal{O}(V + E)$ time**:

1. Start at a valid vertex and traverse unused edges greedily until returning to the start (forming a sub-cycle).
2. If there are vertices along this sub-cycle with remaining untraversed edges, start another walk from that vertex and splice the new cycle into the previous one.
3. Using an explicit traversal stack and reverse post-order construction, this executes in a single clean pass.

def find_eulerian_circuit(num_nodes: int, edges: list[tuple[int, int]]) -> list[int]:
    """Finds an Eulerian path/circuit in a directed graph in O(V + E) time."""
    adj = [[] for _ in range(num_nodes)]
    in_deg = [0] * num_nodes
    out_deg = [0] * num_nodes
    
    for u, v in edges:
        adj[u].append(v)
        out_deg[u] += 1
        in_deg[v] += 1
        
    start = 0
    start_nodes, end_nodes = 0, 0
    for i in range(num_nodes):
        diff = out_deg[i] - in_deg[i]
        if diff == 1:
            start_nodes += 1
            start = i
        elif diff == -1:
            end_nodes += 1
        elif diff != 0:
            return []  # Invalid degrees
            
    if not (start_nodes == 0 and end_nodes == 0) and not (start_nodes == 1 and end_nodes == 1):
        return []
        
    stack = [start]
    path = []
    
    while stack:
        curr = stack[-1]
        if adj[curr]:
            next_node = adj[curr].pop()
            stack.append(next_node)
        else:
            path.append(stack.pop())
            
    path.reverse()
    return path if len(path) == len(edges) + 1 else []

export function findEulerianCircuit(
  numNodes: number,
  edges: [number, number][]
): number[] {
  const adj: number[][] = Array.from({ length: numNodes }, () => []);
  const inDeg = new Int32Array(numNodes);
  const outDeg = new Int32Array(numNodes);

for (const [u, v] of edges) {
adj[u].push(v);
outDeg[u]++;
inDeg[v]++;
}

let start = 0;
let startCount = 0;
let endCount = 0;

for (let i = 0; i < numNodes; i++) {
const diff = outDeg[i] - inDeg[i];
if (diff === 1) {
startCount++;
start = i;
} else if (diff === -1) {
endCount++;
} else if (diff !== 0) {
return [];
}
}

if (
!(startCount === 0 && endCount === 0) &&
!(startCount === 1 && endCount === 1)
) {
return [];
}

const stack: number[] = [start];
const path: number[] = [];

while (stack.length > 0) {
const curr = stack[stack.length - 1];
if (adj[curr].length > 0) {
const nextNode = adj[curr].pop()!;
stack.push(nextNode);
} else {
path.push(stack.pop()!);
}
}

path.reverse();
return path.length === edges.length + 1 ? path : [];
}

#include <vector>
#include <stack>
#include <algorithm>

std::vector<int> findEulerianCircuit(int numNodes, const std::vector<std::pair<int, int>>& edges) {
    std::vector<std::vector<int>> adj(numNodes);
    std::vector<int> inDeg(numNodes, 0), outDeg(numNodes, 0);

    for (const auto& [u, v] : edges) {
        adj[u].push_back(v);
        ++outDeg[u];
        ++inDeg[v];
    }

    int start = 0;
    int startCount = 0, endCount = 0;

    for (int i = 0; i < numNodes; ++i) {
        int diff = outDeg[i] - inDeg[i];
        if (diff == 1) {
            ++startCount;
            start = i;
        } else if (diff == -1) {
            ++endCount;
        } else if (diff != 0) {
            return {};
        }
    }

    if (!(startCount == 0 && endCount == 0) && !(startCount == 1 && endCount == 1)) {
        return {};
    }

    std::vector<int> stack = {start};
    std::vector<int> path;

    while (!stack.empty()) {
        int curr = stack.back();
        if (!adj[curr].empty()) {
            int nextNode = adj[curr].back();
            adj[curr].pop_back();
            stack.push_back(nextNode);
        } else {
            path.push_back(stack.back());
            stack.pop_back();
        }
    }

    std::reverse(path.begin(), path.end());
    if (path.size() != edges.size() + 1) return {};
    return path;
}
