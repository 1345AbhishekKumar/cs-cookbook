While Binary Lifting answers LCA queries in $\mathcal{O}(\log N)$ time, the **Farach-Colton and Bender algorithm** reduces the LCA problem to a static **Range Minimum Query (RMQ)**.

By pairing an Euler Tour recording node visits with a **Sparse Table**, LCA queries are resolved in strictly **$\mathcal{O}(1)$ constant time**!

---

## 1. The Euler Tour Depth Reduction

During a full DFS traversal where we record node $u$ **every time we visit it or return from a child**:

- The total length of this tour is exactly $2N - 1$.
- Let $first[u]$ be the index of node $u$'s first appearance in the tour.
- For any two nodes $u$ and $v$ with $first[u] < first[v]$, the path between them in the tour traverses their Lowest Common Ancestor.
- The LCA is precisely the **node with the minimum depth** in the subarray of the tour between $first[u]$ and $first[v]$!

$$\text{LCA}(u, v) = \text{tour}\Big[ \arg\min_{i \in [first[u], \; first[v]]} \text{depth}[tour[i]] \Big]$$

Because Range Minimum Query on a static array can be answered in $\mathcal{O}(1)$ using a Sparse Table, LCA is answered in $\mathcal{O}(1)$ time!

class LCASparseTable:
    def __init__(self, num_nodes: int, adj: list[list[int]], root: int = 0):
        self.euler_tour = []
        self.depth_tour = []
        self.first_occurrence = [-1] * num_nodes
        
        def dfs(u: int, p: int, d: int):
            self.first_occurrence[u] = len(self.euler_tour)
            self.euler_tour.append(u)
            self.depth_tour.append(d)
            
            for v in adj[u]:
                if v != p:
                    dfs(v, u, d + 1)
                    self.euler_tour.append(u)
                    self.depth_tour.append(d)
                    
        dfs(root, -1, 0)
        
        # Build Sparse Table over depth_tour
        m = len(self.depth_tour)
        k = m.bit_length()
        self.st = [[0] * m for _ in range(k)]
        self.st[0] = list(range(m))
        
        for j in range(1, k):
            half = 1 << (j - 1)
            for i in range(m - (1 << j) + 1):
                idx1 = self.st[j - 1][i]
                idx2 = self.st[j - 1][i + half]
                if self.depth_tour[idx1] <= self.depth_tour[idx2]:
                    self.st[j][i] = idx1
                else:
                    self.st[j][i] = idx2

    def get_lca(self, u: int, v: int) -> int:
        """Computes LCA in strict O(1) time."""
        left = self.first_occurrence[u]
        right = self.first_occurrence[v]
        if left > right:
            left, right = right, left

        length = right - left + 1
        j = length.bit_length() - 1
        idx1 = self.st[j][left]
        idx2 = self.st[j][right - (1 << j) + 1]

        best_idx = idx1 if self.depth_tour[idx1] <= self.depth_tour[idx2] else idx2
        return self.euler_tour[best_idx]

export class LCASparseTable {
  private eulerTour: number[] = [];
  private depthTour: number[] = [];
  private firstOccurrence: Int32Array;
  private st: Int32Array[];

  constructor(numNodes: number, adj: number[][], root: number = 0) {
    this.firstOccurrence = new Int32Array(numNodes).fill(-1);

    const dfs = (u: number, p: number, d: number) => {
      this.firstOccurrence[u] = this.eulerTour.length;
      this.eulerTour.push(u);
      this.depthTour.push(d);

      for (const v of adj[u]) {
        if (v !== p) {
          dfs(v, u, d + 1);
          this.eulerTour.push(u);
          this.depthTour.push(d);
        }
      }
    };

    dfs(root, -1, 0);

    const m = this.depthTour.length;
    const k = Math.floor(Math.log2(m)) + 1;
    this.st = Array.from({ length: k }, () => new Int32Array(m));
    for (let i = 0; i < m; i++) this.st[0][i] = i;

    for (let j = 1; j < k; j++) {
      const half = 1 << (j - 1);
      const limit = m - (1 << j) + 1;
      for (let i = 0; i < limit; i++) {
        const idx1 = this.st[j - 1][i];
        const idx2 = this.st[j - 1][i + half];
        this.st[j][i] = this.depthTour[idx1] <= this.depthTour[idx2] ? idx1 : idx2;
      }
    }
  }

  getLCA(u: number, v: number): number {
    let left = this.firstOccurrence[u];
    let right = this.firstOccurrence[v];
    if (left > right) {
      const temp = left;
      left = right;
      right = temp;
    }

    const len = right - left + 1;
    const j = Math.floor(Math.log2(len));
    const idx1 = this.st[j][left];
    const idx2 = this.st[j][right - (1 << j) + 1];

    const bestIdx = this.depthTour[idx1] <= this.depthTour[idx2] ? idx1 : idx2;
    return this.eulerTour[bestIdx];
  }
}

#include <vector>
#include <bit>
#include <algorithm>

class LCASparseTable {
private:
std::vector<int> eulerTour;
std::vector<int> depthTour;
std::vector<int> firstOccurrence;
std::vector<std::vector<int>> st;

    void dfs(int u, int p, int d, const std::vector<std::vector<int>>& adj) {
        firstOccurrence[u] = static_cast<int>(eulerTour.size());
        eulerTour.push_back(u);
        depthTour.push_back(d);

        for (int v : adj[u]) {
            if (v != p) {
                dfs(v, u, d + 1, adj);
                eulerTour.push_back(u);
                depthTour.push_back(d);
            }
        }
    }

public:
LCASparseTable(int numNodes, const std::vector<std::vector<int>>& adj, int root = 0)
: firstOccurrence(numNodes, -1) {
dfs(root, -1, 0, adj);

        int m = static_cast<int>(depthTour.size());
        int k = std::__lg(m) + 1;
        st.assign(k, std::vector<int>(m));

        for (int i = 0; i < m; ++i) st[0][i] = i;

        for (int j = 1; j < k; ++j) {
            int half = 1 << (j - 1);
            for (int i = 0; i + (1 << j) <= m; ++i) {
                int idx1 = st[j - 1][i];
                int idx2 = st[j - 1][i + half];
                st[j][i] = (depthTour[idx1] <= depthTour[idx2]) ? idx1 : idx2;
            }
        }
    }

    int getLCA(int u, int v) const {
        int left = firstOccurrence[u];
        int right = firstOccurrence[v];
        if (left > right) std::swap(left, right);

        int len = right - left + 1;
        int j = std::__lg(len);
        int idx1 = st[j][left];
        int idx2 = st[j][right - (1 << j) + 1];

        int bestIdx = (depthTour[idx1] <= depthTour[idx2]) ? idx1 : idx2;
        return eulerTour[bestIdx];
    }

};
