**Euler's Totient Function $\phi(N)$** counts the number of positive integers up to $N$ that are **relatively prime (coprime)** to $N$:

$$\phi(N) = \sum_{k=1}^N [\gcd(k, N) == 1]$$

---

## 1. Euler's Product Formula

If the prime factorization of $N$ is $N = p_1^{e_1} p_2^{e_2} \dots p_k^{e_k}$, Euler's product formula states:

$$\phi(N) = N \prod_{p \mid N} \left( 1 - \frac{1}{p} \right)$$

### Properties

1. If $p$ is prime: $\phi(p) = p - 1$.
2. Multiplicative: If $\gcd(a, b) = 1$, then $\phi(a \cdot b) = \phi(a) \cdot \phi(b)$.
3. **Euler's Totient Theorem**: If $\gcd(a, m) = 1$, then $a^{\phi(m)} \equiv 1 \pmod m$.

---

def phi_single(n: int) -> int:
    """Computes phi(n) in O(sqrt(N)) time using prime factorization."""
    res = n
    p = 2
    while p * p <= n:
        if n % p == 0:
            while n % p == 0:
                n //= p
            res -= res // p
        p += 1
    if n > 1:
        res -= res // n
    return res

def totient_sieve(limit: int) -> list[int]:
"""Precomputes phi(1..limit) in O(N log log N) time."""
phi = list(range(limit + 1))
for i in range(2, limit + 1):
if phi[i] == i: # i is prime
for j in range(i, limit + 1, i):
phi[j] -= phi[j] // i
return phi

export function phiSingle(n: number): number {
  let res = n;
  for (let p = 2; p * p <= n; p++) {
    if (n % p === 0) {
      while (n % p === 0) n = Math.floor(n / p);
      res -= Math.floor(res / p);
    }
  }
  if (n > 1) res -= Math.floor(res / n);
  return res;
}

export function totientSieve(limit: number): Int32Array {
  const phi = new Int32Array(limit + 1);
  for (let i = 0; i <= limit; i++) phi[i] = i;

  for (let i = 2; i <= limit; i++) {
    if (phi[i] === i) {
      for (let j = i; j <= limit; j += i) {
        phi[j] -= Math.floor(phi[j] / i);
      }
    }
  }
  return phi;
}
