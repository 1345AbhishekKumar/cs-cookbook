When a dynamic programming recurrence depends linearly on previous $K$ states:

$$f(n) = c_1 f(n-1) + c_2 f(n-2) + \dots + c_K f(n-K)$$

Evaluating $f(N)$ for $N = 10^{18}$ naively takes $\mathcal{O}(N)$ operations, which is completely infeasible.

**Matrix Exponentiation** translates the linear recurrence into a matrix-vector equation:

$$\begin{bmatrix} f(n) \\ f(n-1) \end{bmatrix} = \mathbf{T} \cdot \begin{bmatrix} f(n-1) \\ f(n-2) \end{bmatrix} \implies \begin{bmatrix} f(N) \\ f(N-1) \end{bmatrix} = \mathbf{T}^{N-1} \cdot \begin{bmatrix} f(1) \\ f(0) \end{bmatrix}$$

Using binary matrix exponentiation, $\mathbf{T}^{N-1}$ is evaluated in **$\mathcal{O}(K^3 \log N)$ time**!

---

## 1. Canonical Example: Fibonacci Numbers in $\mathcal{O}(\log N)$

For Fibonacci $F(n) = F(n-1) + F(n-2)$:

$$\begin{bmatrix} F(n) \\ F(n-1) \end{bmatrix} = \begin{bmatrix} 1 & 1 \\ 1 & 0 \end{bmatrix} \begin{bmatrix} F(n-1) \\ F(n-2) \end{bmatrix}$$

def matrix*multiply(a: list[list[int]], b: list[list[int]]) -> list[list[int]]:

base = [row[:] for row in mat]

    while exp > 0:

const MOD = 1_000_000_007n;

function multiply(a: bigint[][], b: bigint[][]): bigint[][] {
  const k = a.length;
  const c: bigint[][] = Array.from({ length: k }, () => Array(k).fill(0n));
  for (let i = 0; i < k; i++) {
    for (let p = 0; p < k; p++) {
      if (a[i][p] === 0n) continue;
      for (let j = 0; j < k; j++) {
        c[i][j] = (c[i][j] + a[i][p] * b[p][j]) % MOD;
      }
    }
  }
  return c;
}

function matrixPower(mat: bigint[][], exp: bigint): bigint[][] {
  const k = mat.length;
  let res: bigint[][] = Array.from({ length: k }, (_, i) =>
    Array.from({ length: k }, (_, j) => (i === j ? 1n : 0n))
  );
  let base = mat.map((row) => [...row]);

  while (exp > 0n) {
    if (exp & 1n) res = multiply(res, base);
    base = multiply(base, base);
    exp >>= 1n;
  }
  return res;
}

export function fibonacciNth(n: number): number {
  if (n === 0) return 0;
  if (n === 1) return 1;
  const T = [
    [1n, 1n],
    [1n, 0n],
  ];
  const TN = matrixPower(T, BigInt(n - 1));
  return Number(TN[0][0]);
}
